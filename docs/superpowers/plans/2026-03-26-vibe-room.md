# Vibe Room Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a showcase website for a musical equipment collection with Payload CMS admin panel and Astro SSG frontend in retrowave neon pink design.

**Architecture:** Monorepo with two apps — Payload CMS 3.x (runs as a Next.js app providing admin UI and API) and Astro 5.x (SSG frontend). Payload stores instruments, categories, and page content in SQLite. Astro fetches data via Payload Local API at build time and generates static HTML. Rebuild triggered by Payload afterChange hooks.

**Tech Stack:** Astro 5.x, Payload CMS 3.x (Next.js 15), SQLite (via @payloadcms/db-sqlite), pnpm workspaces, TypeScript

**Spec:** `docs/superpowers/specs/2026-03-26-vibe-room-design.md`

- [ ] **Step 1: Create rebuild hook**

Create `apps/cms/src/hooks/rebuild.ts`:

```ts
import { exec } from 'child_process'
import path from 'path'
import { fileURLToPath } from 'url'
import type { Payload } from 'payload'

const filename = fileURLToPath(import.meta.url)
const dirname = path.dirname(filename)
const webDir = path.resolve(dirname, '../../../web')

let building = false
let queued = false
let debounceTimer: ReturnType<typeof setTimeout> | null = null

async function runBuild(payload?: Payload): Promise<void> {
  if (building) {
    queued = true
    return
  }

  building = true
  console.log('[rebuild] Starting astro build...')

  return new Promise((resolve) => {
    exec('pnpm build', { cwd: webDir }, async (error, stdout, stderr) => {
      building = false

      if (error) {
        console.error('[rebuild] Build failed:', stderr)
      } else {
        console.log('[rebuild] Build complete.')
        // Update lastBuildAt timestamp
        if (payload) {
          try {
            await payload.updateGlobal({
              slug: 'site-settings',
              data: { lastBuildAt: new Date().toISOString() },
            })
          } catch (e) {
            console.error('[rebuild] Failed to update lastBuildAt:', e)
          }
        }
      }

      if (queued) {
        queued = false
        runBuild(payload).then(resolve)
      } else {
        resolve()
      }
    })
  })
}

export function triggerRebuild(payload?: Payload): void {
  if (debounceTimer) {
    clearTimeout(debounceTimer)
  }

  debounceTimer = setTimeout(() => {
    debounceTimer = null
    runBuild(payload)
  }, 5000)
}
```

- [ ] **Step 6: Create rich text helper**

Payload Lexical stores rich text as JSON. We need to serialize it to HTML for rendering in Astro.

Create `apps/web/src/lib/richtext.ts`:

```ts
import type { SerializedEditorState } from '@payloadcms/richtext-lexical/lexical'

/**
 * Convert Lexical rich text JSON to simple HTML.
 * Handles: paragraphs, headings, lists, bold, italic, links, line breaks.
 */
export function richTextToHtml(content: SerializedEditorState | null | undefined): string {
  if (!content || !content.root || !content.root.children) return ''
  return content.root.children.map(serializeNode).join('')
}

function serializeNode(node: any): string {
  if (!node) return ''

  // Text node
  if (node.type === 'text') {
    let text = escapeHtml(node.text || '')
    if (node.format & 1) text = `<strong>${text}</strong>` // bold
    if (node.format & 2) text = `<em>${text}</em>` // italic
    return text
  }

  // Line break
  if (node.type === 'linebreak') return '<br>'

  // Recursively serialize children
  const children = (node.children || []).map(serializeNode).join('')

  switch (node.type) {
    case 'paragraph':
      return `<p>${children}</p>`
    case 'heading':
      const tag = node.tag || 'h2'
      return `<${tag}>${children}</${tag}>`
    case 'list':
      const listTag = node.listType === 'number' ? 'ol' : 'ul'
      return `<${listTag}>${children}</${listTag}>`
    case 'listitem':
      return `<li>${children}</li>`
    case 'link':
    case 'autolink':
      const url = node.fields?.url || node.url || '#'
      return `<a href="${escapeHtml(url)}" target="_blank" rel="noopener">${children}</a>`
    default:
      return children
  }
}

function escapeHtml(str: string): string {
  return str
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
}

/**
 * Extract plain text from Lexical JSON (for meta descriptions).
 */
export function richTextToPlainText(content: SerializedEditorState | null | undefined): string {
  if (!content || !content.root || !content.root.children) return ''
  return extractText(content.root).trim().slice(0, 160)
}

function extractText(node: any): string {
  if (node.type === 'text') return node.text || ''
  if (node.type === 'linebreak') return ' '
  return (node.children || []).map(extractText).join('')
}
```

```astro
    const mainImg = document.getElementById('gallery-main-img') as HTMLImageElement
    const thumbs = Array.from(document.querySelectorAll<HTMLButtonElement>('.gallery-thumb'))
    const lightbox = document.getElementById('lightbox')!
    const lightboxImg = document.getElementById('lightbox-img') as HTMLImageElement
    const galleryMain = document.getElementById('gallery-main')

    let currentIndex = 0
    const allImages = thumbs.map((t) => ({ src: t.dataset.full!, alt: t.dataset.alt! }))
    // If no thumbs, use main image
    if (allImages.length === 0 && mainImg.src) {
      allImages.push({ src: mainImg.src, alt: mainImg.alt })
    }

    function showThumb(index: number) {
      currentIndex = index
      thumbs.forEach((t, i) => t.classList.toggle('active', i === index))
      mainImg.src = allImages[index].src
      mainImg.alt = allImages[index].alt
    }

    function showLightbox(index: number) {
      currentIndex = index
      lightboxImg.src = allImages[index].src
      lightboxImg.alt = allImages[index].alt
      lightbox.hidden = false
    }

    function navigate(delta: number) {
      const next = (currentIndex + delta + allImages.length) % allImages.length
      currentIndex = next
      lightboxImg.src = allImages[next].src
      lightboxImg.alt = allImages[next].alt
      showThumb(next)
    }

    thumbs.forEach((thumb, i) => {
      thumb.addEventListener('click', () => showThumb(i))
    })

    galleryMain?.addEventListener('click', () => showLightbox(currentIndex))

    lightbox.querySelector('.lightbox-close')?.addEventListener('click', () => { lightbox.hidden = true })
    lightbox.querySelector('.lightbox-prev')?.addEventListener('click', () => navigate(-1))
    lightbox.querySelector('.lightbox-next')?.addEventListener('click', () => navigate(1))
    lightbox.addEventListener('click', (e) => { if (e.target === lightbox) lightbox.hidden = true })

    document.addEventListener('keydown', (e) => {
      if (lightbox.hidden) return
      if (e.key === 'Escape') lightbox.hidden = true
      if (e.key === 'ArrowLeft') navigate(-1)
      if (e.key === 'ArrowRight') navigate(1)
    })

    // Touch swipe support
    let touchStartX = 0
    lightbox.addEventListener('touchstart', (e) => { touchStartX = e.touches[0].clientX }, { passive: true })
    lightbox.addEventListener('touchend', (e) => {
      const diff = e.changedTouches[0].clientX - touchStartX
      if (Math.abs(diff) > 50) navigate(diff > 0 ? -1 : 1)
    })
  }

  initGallery()
</script>
```
